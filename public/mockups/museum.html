Specimen No. 1gems.em
struct Gem {
    const name: String
    const carats: Float
}

const gems = [Gem("Emerald", 2.5), Gem("Ruby", 1.25)]
for gem in gems {
    print("#{gem.name}: #{gem.carats} carats")
}
Observed output:
Emerald: 2.5 carats
Ruby: 1.25 carats
